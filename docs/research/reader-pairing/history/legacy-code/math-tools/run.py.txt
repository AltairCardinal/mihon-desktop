ROOT = Path(__file__).resolve().parents[2]

    READER + name + ".kt"
    for name in (
        "ReaderCanonicalPairingScan", "SharedDualPageRules", "ReaderPairingCertification",

    )

def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def jar(group: str, name: str, version: str) -> Path:
    files = sorted((CACHE / group / name / version).glob(f"*/{name}-{version}.jar"))
    if len(files) != 1:
        raise RuntimeError(f"Expected one cached {group}:{name}:{version}; found {len(files)}. No download attempted.")
    return files[0]


def run(command: list[str], log: Path, timeout: int) -> str:
    with log.open("w", encoding="utf-8", newline="\n") as stream:
        completed = subprocess.run(
            command, cwd=ROOT, stdout=stream, stderr=subprocess.STDOUT,
            env={**os.environ, "PYTHONUTF8": "1", "PYTHONIOENCODING": "utf-8", "PYTHONDONTWRITEBYTECODE": "1"},
            timeout=timeout, check=False, encoding="utf-8",
        )
    output = log.read_text(encoding="utf-8")
    if completed.returncode:
        raise RuntimeError(f"Command failed ({completed.returncode}); inspect {log}\n{output[-6000:]}")
    return output


def main() -> None:
    OUT.mkdir(parents=True, exist_ok=True)
    java = shutil.which("java")
    if not java:
        raise RuntimeError("Java not available")
    compiler = [
        jar("org.jetbrains.kotlin", "kotlin-compiler-embeddable", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-stdlib", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-script-runtime", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-reflect", "1.6.10"),
        jar("org.jetbrains.kotlin", "kotlin-daemon-embeddable", "2.3.20"),
        jar("org.jetbrains.kotlinx", "kotlinx-coroutines-core-jvm", "1.8.0"),
        jar("org.jetbrains", "annotations", "23.0.0"),
    ]
    peripheral = ROOT / "domain/build/libs/domain-jvm.jar"
    if not peripheral.is_file():
        raise RuntimeError("Existing peripheral production jar missing; no Gradle or download attempted")
    main_source = ROOT / "scripts/reader-pairing-math/PairingMathExperiment.kt"
    classes = OUT / "classes"
    classes.mkdir(exist_ok=True)
    compile_classpath = os.pathsep.join(map(str, [compiler[1], peripheral]))
    command = [
        java, "-Xmx768m", "-XX:MaxMetaspaceSize=256m", "-Dfile.encoding=UTF-8",
        "-cp", os.pathsep.join(map(str, compiler)), "org.jetbrains.kotlin.cli.jvm.K2JVMCompiler",
        "-no-stdlib", "-no-reflect", "-jvm-target", "17", "-classpath", compile_classpath,
        "-d", str(classes), *[str(ROOT / source) for source in SOURCES], str(main_source),
    ]
    started = time.monotonic()
    source_hashes = {source: digest(ROOT / source) for source in SOURCES}
    jar_hash = digest(peripheral)

    def stable() -> None:
        if any(digest(ROOT / source) != value for source, value in source_hashes.items()) or digest(peripheral) != jar_hash:
            (OUT / "result.json").write_text('{"status":"SOURCE_CHANGED"}', encoding="utf-8")
            raise RuntimeError("SOURCE_CHANGED: core source or peripheral jar changed during experiment; results invalid")

    manifest = {
        "scope": "algorithm research only; peripheral source/jar equivalence is not established",
        "baselineHead": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True, encoding="utf-8").strip(),
        "sources": source_hashes,
        "experimentSha256": digest(main_source),
        "peripheralJar": {"path": str(peripheral), "sha256": jar_hash},
        "peripheralTypes": ["ReaderChapterId", "ReaderPageId", "ReaderDualPageContentIdentity", "ReaderDualPageIdentity",
                            "ReaderDirection", "ReaderPageSize", "ReaderPairingAdjustment"],
        "peripheralSourceHashes": {source: digest(ROOT / source) for source in (
            READER + "session/ReaderSession.kt", READER + "SharedDualPageSession.kt",
            READER + "ReaderNavigation.kt", READER + "ReaderPairingAdjustment.kt", READER + "ReaderPageModel.kt",
        )},
        "compilerJars": {str(path): digest(path) for path in compiler},
        "compileCommand": command,
        "limits": {"compilerHeapMiB": 768, "compilerMetaspaceMiB": 256, "runHeapMiB": 256,
                   "compileTimeoutSeconds": 60, "runTimeoutSeconds": 60, "maxPages": 8, "maxCases": 10000},
    }
    (OUT / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    (OUT / "result.json").write_text('{"status":"RUNNING"}', encoding="utf-8")
    javap = Path(java).with_name("javap.exe" if os.name == "nt" else "javap")
    if not javap.is_file():
        raise RuntimeError("javap missing; cannot record peripheral signatures")
    run([
        str(javap), "-classpath", str(peripheral),
        *["mihon.domain.reader." + name for name in manifest["peripheralTypes"]
          if name not in ("ReaderChapterId", "ReaderPageId")],
        "mihon.domain.reader.session.ReaderChapterId", "mihon.domain.reader.session.ReaderPageId",
    ], OUT / "peripheral-signatures.log", 15)
    variants = {}
    certifier = ROOT / (READER + "ReaderPairingCertification.kt")
    with certifier.open(encoding="utf-8", newline="") as source_stream:
        original = source_stream.read()
    needle = "                    add(ReaderPairingScanCursor(first + 1, false))"
    if original.count(needle) != 1:
        raise RuntimeError("Expected exactly one shifted-entry source line; current source is unsupported")
    newline = "\r\n" if "\r\n" in original else "\n"
    replacement = newline.join([
        "                    if (pages[first - 1] !in settings.forcedSinglePages &&",
        "                        pages[first] !in settings.forcedSinglePages",
        "                    ) {",
        "                        add(ReaderPairingScanCursor(first + 1, false))",
        "                    }",
    ])
    candidate_source = OUT / "candidate-source" / "ReaderPairingCertification.kt"
    candidate_source.parent.mkdir(exist_ok=True)
    with candidate_source.open("w", encoding="utf-8", newline="") as candidate_stream:
        candidate_stream.write(original.replace(needle, replacement))
    patch = "".join(difflib.unified_diff(
        original.splitlines(keepends=True), original.replace(needle, replacement).splitlines(keepends=True),
        fromfile="production/ReaderPairingCertification.kt", tofile="isolated-candidate/ReaderPairingCertification.kt",
    ))
    (OUT / "candidate.patch").write_text(patch, encoding="utf-8")
    manifest["isolatedCandidateSha256"] = digest(candidate_source)
    manifest["isolatedPatch"] = patch
    (OUT / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    for variant in ("baseline", "candidate"):
        stable()
        variant_classes = OUT / (variant + "-classes")
        variant_classes.mkdir(exist_ok=True)
        variant_command = command.copy()
        variant_command[variant_command.index("-d") + 1] = str(variant_classes)
        if variant == "candidate":
            variant_command[variant_command.index(str(certifier))] = str(candidate_source)
        print(f"Compile {variant}; timeout=60s; log={OUT / (variant + '-compile.log')}", flush=True)
        run(variant_command, OUT / (variant + "-compile.log"), 60)
        stable()
        print(f"Run {variant}; timeout=60s; log={OUT / (variant + '-run.log')}", flush=True)
        result = run([
            java, "-Xmx256m", "-XX:MaxMetaspaceSize=128m", "-Dfile.encoding=UTF-8", "-cp",
            os.pathsep.join(map(str, [variant_classes, compiler[1], peripheral])),
            "mihon.domain.reader.PairingMathExperimentKt", variant,
        ], OUT / (variant + "-run.log"), 60)
        stable()
        variants[variant] = json.loads(result)
        (OUT / (variant + ".json")).write_text(json.dumps(variants[variant], indent=2), encoding="utf-8")
    baseline = {row["key"]: row for row in variants["baseline"]["pruning"]}
    candidate = {row["key"]: row for row in variants["candidate"]["pruning"]}
    assert baseline.keys() == candidate.keys()
    transitions: dict[str, int] = {}
    for key, old in baseline.items():
        new = candidate[key]
        assert old["oracle"] == new["oracle"], key
        if old["result"]["status"] == "Certified":
            assert new["result"] == old["result"], key
        transition = old["result"]["status"] + "->" + new["result"]["status"]
        transitions[transition] = transitions.get(transition, 0) + 1
    strict_key = "5-LTR-2-AUTO-1"
    persisted_key = "6-LTR-2-AUTO-1"
    decoded = {
        "status": "PASS", "sourceStable": True, "baselineHead": manifest["baselineHead"],
        "casesPerVariant": variants["baseline"]["cases"], "pairedPruningCases": len(baseline),
        "transitions": transitions,
        "strictFixture": {"baseline": baseline[strict_key], "candidate": candidate[strict_key]},
        "validPersistedFixture": {"baseline": baseline[persisted_key], "candidate": candidate[persisted_key],
                                  "zeroBasedAnalysisOrder": [0, 3, 4, 2, 1, 5], "forcedSinglePage": 1,
                                  "target": 4, "ownerCapacity": 2},
        "persistedValidPruningCases": sum(row["persistedRecordValid"] for row in baseline.values()),
        "domain": {"pageCounts": [5, 6, 7, 8], "directions": ["LTR", "RTL"], "capacities": [2, 16],
                   "openingModes": ["AUTO", "FIRST_SINGLE", "FIRST_PAIR"],
                   "forceMasks": "all 32 subsets of last five pages; earlier pages unforced",
                   "sides": "leading side at page n-4; all others UNKNOWN; all from physical pixel functions",
                   "geometry": "all portraits in pruning family; 4 separate lower-bound portrait/spread cases",
                   "target": "n-2", "analysisOrder": "earlier prefix, then local indices 2,3,1,0,4",
                   "manualPairStarts": [], "automatic": True,
                   "capacityCaveat": "capacity 2 is supported owner state; not evidence of default-16 five-page eviction"},
        "elapsedSeconds": round(time.monotonic() - started, 3),
    }
    stable()
    manifest["sourcesAfter"] = {source: digest(ROOT / source) for source in SOURCES}
    manifest["peripheralJarAfterSha256"] = digest(peripheral)
    manifest["sourceStable"] = True
    (OUT / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    (OUT / "result.json").write_text(json.dumps(decoded, indent=2), encoding="utf-8")
    print(json.dumps({"status": "PASS", "pairedCases": len(baseline), "transitions": transitions,
                      "result": str(OUT / "result.json")}), flush=True)

    stable()
    certifier = ROOT / (READER + "ReaderPairingCertification.kt")

    needle = "val isolatedSpread = targetFact.state == ReaderPageFactState.SPREAD"

        raise RuntimeError("Unsupported isolatedSpread condition; expected exactly one source occurrence")

    candidate_source = output / "forced-candidate-source" / certifier.name

        candidate_stream.write(patched)

        original.splitlines(keepends=True), patched.splitlines(keepends=True),
        fromfile="current-production/ReaderPairingCertification.kt",
        tofile="round2-isolated-forced/ReaderPairingCertification.kt",

    run(candidate_compile, output / "forced-candidate-compile.log", 60)
    stable()
    candidate_run = run_command.copy()
    candidate_run[candidate_run.index("-cp") + 1] = os.pathsep.join(map(str, [candidate_classes, compiler[1], peripheral]))
    print(f"Run round2 forced candidate; timeout=60s; log={output / 'forced-candidate.log'}", flush=True)
    forced_candidate = json.loads(run(candidate_run + ["forced-candidate"], output / "forced-candidate.log", 60))
    stable()
    transitions: dict[str, int] = {}
    assert forced_baseline["cases"] == forced_candidate["cases"] == 96
    for old, new in zip(forced_baseline["records"], forced_candidate["records"], strict=True):
        assert old["key"] == new["key"] and old["oracle"] == new["oracle"]
        if old["status"] == "Certified":
            assert new["status"] == "Certified"
        transition = old["status"] + "->" + new["status"]
        transitions[transition] = transitions.get(transition, 0) + 1
    for name, data in (("forced-baseline.json", forced_baseline), ("forced-candidate.json", forced_candidate)):
        (output / name).write_text(json.dumps(data, indent=2), encoding="utf-8")
    result["forcedCasesPerVariant"] = 96
    result["forcedTransitions"] = transitions
    core_key = "GRAY-LTR-AUTO-true-true"
    result["forcedCore"] = {
        "baseline": next(row for row in forced_baseline["records"] if row["key"] == core_key),
        "candidate": next(row for row in forced_candidate["records"] if row["key"] == core_key),
    }
    result["elapsedCompileAndResearchSeconds"] = round(time.monotonic() - started, 3)
    result["sourceStable"] = True
    manifest["hashesAfter"] = {str(path): digest(path) for path in watched}
    manifest["sourceStable"] = True
    (output / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    (output / "result.json").write_text(json.dumps(result, indent=2), encoding="utf-8")
    print(json.dumps({"status": result["status"], "subjectQueries": result["subjectQueries"],
                      "probeQueries": result["probeQueries"], "result": str(output / "result.json")}), flush=True)


def round3() -> None:
    """Compare two independent demand-reduction candidates against the current production baseline."""
    output = OUT / "round3"
    output.mkdir(parents=True, exist_ok=True)
    java = shutil.which("java")
    if not java:
        raise RuntimeError("Java not available")
    compiler = [
        jar("org.jetbrains.kotlin", "kotlin-compiler-embeddable", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-stdlib", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-script-runtime", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-reflect", "1.6.10"),
        jar("org.jetbrains.kotlin", "kotlin-daemon-embeddable", "2.3.20"),
        jar("org.jetbrains.kotlinx", "kotlinx-coroutines-core-jvm", "1.8.0"),
        jar("org.jetbrains", "annotations", "23.0.0"),
    ]
    peripheral = ROOT / "domain/build/libs/domain-jvm.jar"
    research = ROOT / "scripts/reader-pairing-math/PairingDemandExperiment.kt"
    watched = [*[ROOT / name for name in SOURCES], peripheral, research, Path(__file__).resolve()]
    hashes = {str(path): digest(path) for path in watched}

    def stable() -> None:
        if any(digest(Path(path)) != value for path, value in hashes.items()):
            (output / "result.json").write_text('{"status":"SOURCE_CHANGED"}', encoding="utf-8")
            raise RuntimeError("SOURCE_CHANGED: round3 inputs changed")

    scan_name = READER + "ReaderCanonicalPairingScan.kt"
    cert_name = READER + "ReaderPairingCertification.kt"
    originals = {name: (ROOT / name).read_text(encoding="utf-8") for name in (scan_name, cert_name)}

    def once(source: str, old: str, new: str) -> str:
        if source.count(old) != 1:
            raise RuntimeError("Unsupported production patch anchor: " + old[:90])
        return source.replace(old, new)

    scan_a = once(originals[scan_name], "    private val distance: (ReaderPageId, ReaderPageId) -> Double?,\n",
                  "    private val distance: (ReaderPageId, ReaderPageId) -> Double?,\n    private val distancesNonNegative: Boolean = false,\n")
    scan_a = once(scan_a, "        val laterDistance = distance(pages[i + 1], pages[i + 2])",
                  "        if (distancesNonNegative && nearDistance == 0.0) return false\n        val laterDistance = distance(pages[i + 1], pages[i + 2])")
    cert_a = once(originals[cert_name], "            ::resolvedSide,\n        ) { a, b ->",
                  "            ::resolvedSide,\n            distance = { a, b ->")
    cert_a = once(cert_a, "        }\n        data class State(val index: Int, val opening: Boolean, val reserved: Boolean)",
                  "            },\n            distancesNonNegative = true,\n        )\n        data class State(val index: Int, val opening: Boolean, val reserved: Boolean)")
    cert_b = once(originals[cert_name], "        var first = if (isolatedSpread) targetIndex else maxOf(0, targetIndex - 2)",
                  "        val leadingTarget = !isolatedSpread && settings.automatic && targetIndex > 0 &&\n" +
                  "            targetIndex - 1 !in settings.manualPairStarts &&\n" +
                  "            !(targetIndex == 1 && settings.openingMode == ReaderChapterOpeningMode.FIRST_PAIR) &&\n" +
                  "            resolvedSide(targetIndex) == if (settings.direction == ReaderDirection.RTL) ReaderPageSideHint.RIGHT else ReaderPageSideHint.LEFT\n" +
                  "        val directEntry = isolatedSpread || leadingTarget\n" +
                  "        var first = if (directEntry) targetIndex else maxOf(0, targetIndex - 2)")
    cert_b = once(cert_b, "            } else if (isolatedSpread) {",
                  "            } else if (isolatedSpread || (leadingTarget && first == targetIndex)) {")
    patches = {"baseline": {}, "A": {scan_name: scan_a, cert_name: cert_a}, "B": {cert_name: cert_b}}
    prior_focused = None
    if (output / "result.json").is_file() and (output / "manifest.json").is_file():
        prior_result = json.loads((output / "result.json").read_text(encoding="utf-8"))
        prior_manifest = json.loads((output / "manifest.json").read_text(encoding="utf-8"))
        if prior_result.get("status") == "PASS":
            prior_focused = {"baselineHead": prior_manifest["baselineHead"],
                             "hashesBefore": prior_manifest["hashesBefore"], "result": prior_result,
                             "scopeNote": "First focused pass; did not cover two-opening disagreement fallback"}
    manifest = {
        "scope": "round3 physical owner/scan/certifier algorithm experiment; no platform/performance claim",
        "baselineHead": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True, encoding="utf-8").strip(),
        "hashesBefore": hashes, "compilerJars": {str(path): digest(path) for path in compiler}, "commands": {},
        "limits": {"compilerHeapMiB": 768, "compilerMetaspaceMiB": 256, "runHeapMiB": 256, "timeoutSeconds": 60},
        "priorFocusedPass": prior_focused,
    }
    (output / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    (output / "result.json").write_text('{"status":"RUNNING"}', encoding="utf-8")
    variants = {}
    started = time.monotonic()
    for variant, changed in patches.items():
        stable()
        variant_dir = output / variant
        variant_dir.mkdir(exist_ok=True)
        source_paths = {}
        patch = ""
        for name, source in changed.items():
            destination = variant_dir / Path(name).name
            destination.write_text(source, encoding="utf-8", newline="\n")
            source_paths[name] = destination
            patch += "".join(difflib.unified_diff(originals[name].splitlines(keepends=True), source.splitlines(keepends=True),
                                                 fromfile="production/" + Path(name).name, tofile=variant + "/" + Path(name).name))
        (variant_dir / "candidate.patch").write_text(patch, encoding="utf-8")
        classes = variant_dir / "classes"
        classes.mkdir(exist_ok=True)
        command = [java, "-Xmx768m", "-XX:MaxMetaspaceSize=256m", "-Dfile.encoding=UTF-8", "-cp",
                   os.pathsep.join(map(str, compiler)), "org.jetbrains.kotlin.cli.jvm.K2JVMCompiler", "-no-stdlib", "-no-reflect",
                   "-jvm-target", "17", "-classpath", os.pathsep.join(map(str, [compiler[1], peripheral])), "-d", str(classes),
                   *[str(source_paths.get(name, ROOT / name)) for name in SOURCES], str(research)]
        run_command = [java, "-Xmx256m", "-XX:MaxMetaspaceSize=128m", "-Dfile.encoding=UTF-8", "-cp",
                       os.pathsep.join(map(str, [classes, compiler[1], peripheral])), "mihon.domain.reader.PairingDemandExperimentKt", variant]
        manifest["commands"][variant] = {"compile": command, "run": run_command,

def round5() -> None:
    """Measure the implemented production candidates against the recorded baseline.

    The baseline is the committed production source, the candidates are the
    working tree. Two extra builds apply one candidate at a time, so the
    independent contribution of each candidate is visible, and one build applies
    both research patches to prove that the implementation equals them.
    """
    output = OUT / "round5"
    output.mkdir(parents=True, exist_ok=True)
    java = shutil.which("java")
    if not java:
        raise RuntimeError("Java not available")
    compiler = [
        jar("org.jetbrains.kotlin", "kotlin-compiler-embeddable", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-stdlib", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-script-runtime", "2.3.20"),
        jar("org.jetbrains.kotlin", "kotlin-reflect", "1.6.10"),
        jar("org.jetbrains.kotlin", "kotlin-daemon-embeddable", "2.3.20"),
        jar("org.jetbrains.kotlinx", "kotlinx-coroutines-core-jvm", "1.8.0"),
        jar("org.jetbrains", "annotations", "23.0.0"),
    ]
    peripheral = ROOT / "domain/build/libs/domain-jvm.jar"
    if not peripheral.is_file():
        raise RuntimeError("Existing peripheral production jar missing; no Gradle or download attempted")
    research = ROOT / "scripts/reader-pairing-math/PairingImplementationExperiment.kt"
    counters = ROOT / "scripts/reader-pairing-math/PairingQueryCounters.kt"
    watched = [*[ROOT / name for name in SOURCES], peripheral, research, counters, Path(__file__).resolve()]
    hashes = {str(path): digest(path) for path in watched}

    def stable() -> None:
        if any(digest(Path(path)) != value for path, value in hashes.items()):
            (output / "result.json").write_text('{"status":"SOURCE_CHANGED"}', encoding="utf-8")
            raise RuntimeError("SOURCE_CHANGED: round5 inputs changed")

    scan_name = READER + "ReaderCanonicalPairingScan.kt"
    cert_name = READER + "ReaderPairingCertification.kt"
    owner_name = READER + "ReaderSpreadSamplesOwner.kt"

    def once(source: str, old: str, new: str) -> str:
        if source.count(old) != 1:
            raise RuntimeError("Unsupported round5 patch anchor: " + old[:90])
        return source.replace(old, new)

    def committed() -> dict[str, str]:
        return {
            name: subprocess.check_output(["git", "show", f"HEAD:{name}"], cwd=ROOT).decode("utf-8")
            for name in SOURCES
        }

    def working() -> dict[str, str]:
        return {name: (ROOT / name).read_text(encoding="utf-8") for name in SOURCES}

    # The research candidate patches, unchanged, applied to the committed source.
    def candidate_c(sources: dict[str, str]) -> dict[str, str]:
        scan = once(sources[scan_name], "    private val distance: (ReaderPageId, ReaderPageId) -> Double?,\n",
                    "    private val distance: (ReaderPageId, ReaderPageId) -> Double?,\n"
                    "    private val knownLaterNoMatch: ((ReaderPageId, ReaderPageId) -> Boolean)? = null,\n")
        scan = once(scan,
                    "        val nearDistance = distance(pages[i], pages[i + 1])\n"
                    "        val laterDistance = distance(pages[i + 1], pages[i + 2])",
                    "        val hintedLater = if (knownLaterNoMatch?.invoke(pages[i + 1], pages[i + 2]) == true) {\n"
                    "            val observed = distance(pages[i + 1], pages[i + 2])\n"
                    "            if (observed == null) return false\n            observed\n        } else null\n"
                    "        val nearDistance = distance(pages[i], pages[i + 1])\n"
                    "        val laterDistance = hintedLater ?: distance(pages[i + 1], pages[i + 2])")
        cert = once(sources[cert_name], "            ::resolvedSide,\n        ) { a, b ->",
                    "            ::resolvedSide,\n            distance = { a, b ->")
        cert = once(cert,
                    "        }\n        data class State(val index: Int, val opening: Boolean, val reserved: Boolean)",
                    "            },\n            knownLaterNoMatch = { a, b ->\n"
                    "                val evidence = owner.edgeAnalysis(a, b, settings.direction)\n"
                    "                evidence is ReaderEdgeAnalysisEvidence.Computed && evidence.score == null\n"
                    "            },\n        )\n        data class State(val index: Int, val opening: Boolean, val reserved: Boolean)")
        return {**sources, scan_name: scan, cert_name: cert}

    def candidate_d(sources: dict[str, str]) -> dict[str, str]:
        old_entries = """                    for (opening in listOf(true, false)) {
                        add(ReaderPairingScanCursor(first, opening))
                        add(ReaderPairingScanCursor(first, opening, first))
                    }"""
        new_entries = """                    for (opening in if (first == 1) listOf(true) else listOf(true, false)) {
                        add(ReaderPairingScanCursor(first, opening))
                        if (first > 1 && pages[first - 1] !in settings.forcedSinglePages &&
                            pages[first] !in settings.forcedSinglePages
                        ) {
                            add(ReaderPairingScanCursor(first, opening, first))
                        }
                    }"""
        return {**sources, cert_name: once(sources[cert_name], old_entries, new_entries)}

    # Research-only instrumentation: it counts owner queries and hint hits.
    def instrument(sources: dict[str, str]) -> dict[str, str]:
        owner = once(sources[owner_name],
                     "    fun edgeAnalysis(\n        first: ReaderPageId,\n        second: ReaderPageId,\n"
                     "        direction: ReaderDirection,\n    ): ReaderEdgeAnalysisEvidence {\n",
                     "    fun edgeAnalysis(\n        first: ReaderPageId,\n        second: ReaderPageId,\n"
                     "        direction: ReaderDirection,\n    ): ReaderEdgeAnalysisEvidence {\n"
                     "        PairingQueryCounters.ownerEdgeQueries++\n")
        cert = sources[cert_name]
        hint_anchor = ("            knownLaterNoMatch = { a, b ->\n"
                       "                val evidence = owner.edgeAnalysis(a, b, settings.direction)\n"
                       "                evidence is ReaderEdgeAnalysisEvidence.Computed && evidence.score == null\n"
                       "            },")
        if hint_anchor in cert:
            cert = once(cert, hint_anchor,
                        "            knownLaterNoMatch = { a, b ->\n"
                        "                PairingQueryCounters.hintQueries++\n"
                        "                val evidence = owner.edgeAnalysis(a, b, settings.direction)\n"
                        "                val hit = evidence is ReaderEdgeAnalysisEvidence.Computed && evidence.score == null\n"
                        "                if (hit) PairingQueryCounters.hintHits++\n"
                        "                hit\n            },")
        return {**sources, owner_name: owner, cert_name: cert}

    base = committed()
    variants = {
        "before": instrument(base),
        "afterC": instrument(candidate_c(base)),
        "afterD": instrument(candidate_d(base)),
        "afterCD": instrument(candidate_d(candidate_c(base))),
        "after": instrument(working()),
    }
    manifest = {
        "scope": "round5 implementation measurement; no product/performance claim",
        "baselineHead": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True,
                                                encoding="utf-8").strip(),
        "hashesBefore": hashes, "compilerJars": {str(path): digest(path) for path in compiler}, "commands": {},
        "limits": {"compilerHeapMiB": 768, "compilerMetaspaceSizeMiB": 256, "runHeapMiB": 256, "timeoutSeconds": 60},
    }
    (output / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    (output / "result.json").write_text('{"status":"RUNNING"}', encoding="utf-8")
    results = {}
    started = time.monotonic()
    for variant, sources in variants.items():
        stable()
        directory = output / variant
        directory.mkdir(exist_ok=True)
        paths = {}
        for name, source in sources.items():
            destination = directory / Path(name).name
            destination.write_text(source, encoding="utf-8", newline="\n")
            paths[name] = destination
        classes = directory / "classes"
        classes.mkdir(exist_ok=True)
        compile_command = [java, "-Xmx768m", "-XX:MaxMetaspaceSize=256m", "-Dfile.encoding=UTF-8", "-cp",
                           os.pathsep.join(map(str, compiler)), "org.jetbrains.kotlin.cli.jvm.K2JVMCompiler", "-no-stdlib",
                           "-no-reflect", "-jvm-target", "17", "-classpath",
                           os.pathsep.join(map(str, [compiler[1], peripheral])), "-d", str(classes),
                           *[str(paths[name]) for name in SOURCES], str(research), str(counters)]
        run_command = [java, "-Xmx256m", "-XX:MaxMetaspaceSize=128m", "-Dfile.encoding=UTF-8", "-cp",
                       os.pathsep.join(map(str, [classes, compiler[1], peripheral])),
                       "mihon.domain.reader.PairingImplementationExperimentKt"]
        manifest["commands"][variant] = {"compile": compile_command, "run": run_command}
        print(f"Compile round5 {variant}; timeout=60s; log={directory / 'compile.log'}", flush=True)
        run(compile_command, directory / "compile.log", 60)
        stable()
        print(f"Run round5 {variant}; timeout=60s; log={directory / 'run.log'}", flush=True)
        results[variant] = json.loads(run(run_command, directory / "run.log", 60))
        stable()
        (directory / "result.json").write_text(json.dumps(results[variant], indent=2), encoding="utf-8")

    def index(variant: str) -> dict[str, dict]:
        return {row["key"]: row for row in results[variant]["records"]}

    before = index("before")
    report = {"status": "PASS", "casesPerVariant": len(before), "variants": {}, "witnesses": {}}
    for variant in ("afterC", "afterD", "afterCD", "after"):
        rows = index(variant)
        assert before.keys() == rows.keys()
        transitions: dict[str, int] = {}
        read_delta: dict[str, int] = {}
        needs_delta: dict[str, int] = {}
        query_delta = 0
        hint_queries = 0
        hint_hits = 0
        for key, old in before.items():
            new = rows[key]
            assert new["oracle"] == old["oracle"], key
            if old["status"] == "Certified":
                assert new["status"] == "Certified", key
            assert new["failures"] == old["failures"], key
            assert {json.dumps(x, sort_keys=True) for x in new["needs"]} <= {
                json.dumps(x, sort_keys=True) for x in old["needs"]}, key
            if key.startswith("R3-"):
                assert new["readKeys"] == old["readKeys"], key
            transition = old["status"] + "->" + new["status"]
            transitions[transition] = transitions.get(transition, 0) + 1
            read_delta[str(new["readCount"] - old["readCount"])] = read_delta.get(
                str(new["readCount"] - old["readCount"]), 0) + 1
            needs_delta[str(len(new["needs"]) - len(old["needs"]))] = needs_delta.get(
                str(len(new["needs"]) - len(old["needs"])), 0) + 1
            query_delta += new["ownerQueries"] - old["ownerQueries"]
            hint_queries += new["hintQueries"]
            hint_hits += new["hintHits"]
        report["variants"][variant] = {
            "transitions": transitions,
            "readCountDeltaHistogram": read_delta,
            "needsCountDeltaHistogram": needs_delta,
            "ownerQueryDelta": query_delta,
            "hintQueries": hint_queries,
            "hintHits": hint_hits,
            "hintHitRate": round(hint_hits / hint_queries, 4) if hint_queries else None,
            "readCountTotal": sum(row["readCount"] for row in rows.values()),
            "needsTotal": sum(len(row["needs"]) for row in rows.values()),
            "ownerQueriesTotal": sum(row["ownerQueries"] for row in rows.values()),
        }
    # The shipped production must equal the research D patch. C stays a measured
    # candidate: its research patch is built here, but it is not in production.
    implemented = index("after")
    patched = index("afterD")
    equivalent = True
    for key, row in implemented.items():
        other = patched[key]
        if (row["status"], row["needs"], row["failures"], row["readKeys"], row["ownerQueries"], row["hintQueries"],
                row["hintHits"]) != (other["status"], other["needs"], other["failures"], other["readKeys"],
                                     other["ownerQueries"], other["hintQueries"], other["hintHits"]):
            equivalent = False
            print(f"NOT EQUIVALENT: {key}", flush=True)
    report["implementationEqualsResearchDPatch"] = equivalent
    report["candidateCIsNotShipped"] = True
    for name, key in (("C", "C-16-POSITIVE-NO_MATCH-RTL-AUTO-true"),
                      ("D", "D-16-FORCE2-POSITIVE-RTL-FIRST_PAIR-true")):
        report["witnesses"][name] = {
            "key": key,
            "before": {k: before[key][k] for k in ("status", "needs", "readCount", "ownerQueries", "hintQueries",
                                                   "hintHits", "oracle")},
            "after": {k: implemented[key][k] for k in ("status", "needs", "readCount", "ownerQueries", "hintQueries",
                                                      "hintHits", "oracle")},
        }
    report["baselineReadCountTotal"] = sum(row["readCount"] for row in before.values())
    report["baselineNeedsTotal"] = sum(len(row["needs"]) for row in before.values())
    report["baselineOwnerQueriesTotal"] = sum(row["ownerQueries"] for row in before.values())
    report["elapsedCompileAndResearchSeconds"] = round(time.monotonic() - started, 3)
    report["sourceStable"] = True
    manifest["hashesAfter"] = {str(path): digest(path) for path in watched}
    manifest["sourceStable"] = True
    (output / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    (output / "result.json").write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(json.dumps(report), flush=True)
